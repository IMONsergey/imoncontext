    screen: "intro",
    sectionIndex: 0,

  });

  let state = loadState();

  resetButton.addEventListener("click", () => {
    if (!window.confirm("Удалить все ответы и начать диагностику заново?")) return;
    localStorage.removeItem(STORAGE_KEY);
    state = initialState();
    render();
    toast("Ответы удалены");
  });

  function loadState() {
    try {
      const saved = JSON.parse(localStorage.getItem(STORAGE_KEY));
      if (saved && saved.version === data.version) return { ...initialState(), ...saved };
    } catch (error) {
      console.warn("Не удалось восстановить прогресс", error);
    }
    return initialState();
  }

  function saveState() {
    state.updatedAt = new Date().toISOString();
    localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
  }

  function escapeHtml(value) {
    return String(value ?? "")
      .replaceAll("&", "&amp;")
      .replaceAll("<", "&lt;")
      .replaceAll(">", "&gt;")
      .replaceAll('"', "&quot;")
      .replaceAll("'", "&#039;");
  }

  function visibleQuestions(sectionId = null) {
    return data.questions.filter((question) => {
      if (sectionId && question.section !== sectionId) return false;
      return evaluateCondition(question.showIf);
    });
  }

  function evaluateCondition(condition) {
    if (!condition) return true;
    const answer = state.answers[condition.questionId];
    if (condition.exists) return answer !== undefined && answer !== null && answer !== "";
    if (condition.equals !== undefined) return answer === condition.equals;
    if (condition.includes !== undefined) return Array.isArray(answer) && answer.includes(condition.includes);
    if (condition.min !== undefined) return Number(answer) >= condition.min;
    return true;
  }

  function hasAnswer(question) {
    const answer = state.answers[question.id];
    if (question.type === "multi") return Array.isArray(answer) && answer.length > 0;
    if (question.type === "scale") return Number.isFinite(Number(answer));
    if (question.type === "number") return answer !== undefined && answer !== null && answer !== "";
    return typeof answer === "string" ? answer.trim().length > 0 : answer !== undefined && answer !== null;
  }

  function sectionCompletion(sectionId) {
    const required = visibleQuestions(sectionId).filter((q) => q.required);
    const answered = required.filter(hasAnswer);
    return { required: required.length, answered: answered.length, complete: required.length === answered.length };
  }

  function overallCompletion() {
    const required = visibleQuestions().filter((q) => q.required);
    const answered = required.filter(hasAnswer);
    return { required: required.length, answered: answered.length, percent: required.length ? Math.round((answered.length / required.length) * 100) : 0 };
  }
